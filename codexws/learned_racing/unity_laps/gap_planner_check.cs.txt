// Run with the managed Unity math assemblies and the compiled LapLab experiment DLL.
using System;
using System.Collections.Generic;
using LapLab;
public static class CheckVisibleGaps {
    static Sight Cone(float z,float x) { return new Sight{z=z,x=x,history=1}; }
    static void Require(bool condition,string message){if(!condition)throw new Exception(message);}
    public static void Main(){
        var cfg=new GapConfig{openings=true,paceScale=1,lateralRate=1.8f,responseTime=.2f,
            padding=.12f,speedFractions=new[]{1f,.7f,.45f}};
        var planner=new GapPlanner(cfg);var line=new float[4];
        var cones=new List<Sight>{Cone(5.4f,-2.35f),Cone(5.4f,1.0f),Cone(11.8f,-1.8f),Cone(11.8f,.45f)};
        Require(planner.Plan(line,cones,4.4f,4.45f,0,0,.2f),"No path");
        var d=planner.Decision;
        Require(d.crossings.Length==2,"The two rows were merged");
        Require(d.crossings[0][3]>=-1.51f && d.crossings[0][3]<=.161f,"Unnecessary detour around first opening");
        Require(d.crossings[1][3]<=-.389f && d.crossings[1][3]>=-.961f,"Missed offset second opening");
        Require(d.speed>=4,"Unexpected speed loss for wide reachable openings");
        Console.WriteLine("crossing lateral coordinates {0:0.000}, {1:0.000} m at {2:0.000} m/s",
                          d.crossings[0][3],d.crossings[1][3],d.speed);
        var clear=new GapPlanner(cfg);Require(!clear.Plan(line,new List<Sight>(),4,4,0,0,.2f),"Changed same-pace free-road control");
    }
}
