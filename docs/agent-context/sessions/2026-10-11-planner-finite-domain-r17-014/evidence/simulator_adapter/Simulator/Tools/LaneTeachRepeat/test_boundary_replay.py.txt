"""Boundary replay routing contracts, using explicit core/transport doubles only."""
import contextlib
import io
import json
from pathlib import Path
import sys
from types import SimpleNamespace
import unittest
from unittest.mock import Mock, patch

import numpy as np

import replay_observed_keyframe as replay
import test_replay_plane_calibration as plane
from motion_channel import IMU_PROTOCOL, PROTOCOL


class BoundaryReplayRouting(unittest.TestCase):
    setUp = plane.PlaneReplayRouting.setUp
    execute = plane.PlaneReplayRouting.execute
    assert_invalid_before_core = plane.PlaneReplayRouting.assert_invalid_before_core

    def extractor(self, version=2):
        observation = object()
        extract = Mock(return_value=observation)
        self.modules['lane_mapping_lya_reference.paint_geometry'] = SimpleNamespace(
            extract_boundary_observation=extract, BOUNDARY_VERSION=version,
            TOPOLOGY_REPRESENTATION=replay.BOUNDARY_TOPOLOGY if version==2 else None)
        return observation, extract

    def assert_boundary_call(self, observation, extract):
        args, kwargs = extract.call_args
        np.testing.assert_array_equal(args[0], np.zeros((2, 2), dtype=np.uint8))
        self.assertIs(args[1], self.lookup)
        self.assertEqual(tuple(args[2]), self.sample.attitude.up_body)
        self.assertEqual(args[3], 1_020_000_000)
        self.assertEqual(kwargs, {'threshold': 127})
        self.assertEqual(self.engine.process.call_args.kwargs, {
            'motion_source_stamp_ns': self.sample.stamp_ns,
            'boundary_observation': observation})

    def test_old_fixture_stays_legacy_despite_new_modules(self):
        _, extract = self.extractor()
        report = self.execute(motion_factor=True)
        extract.assert_not_called()
        self.assertFalse(report['candidate_boundary_override'])
        self.assertEqual(report['feature_representation'], 'legacy_filled_consensus')
        self.assertEqual(self.engine.process.call_args.kwargs,
                         {'motion_source_stamp_ns': self.sample.stamp_ns})

    def test_explicit_candidate_uses_same_full_mask_lookup_and_causal_up(self):
        observation, extract = self.extractor()
        report = self.execute(motion_factor=True, candidate_evaluation=True, candidate_boundary=True)
        self.assert_boundary_call(observation, extract)
        self.assertTrue(report['candidate_boundary_override'])
        self.assertEqual(report['candidate_boundary_version'],2)
        self.assertIsNone(report['recorded_boundary_version'])
        self.assertTrue(report['changed_model'])
        self.assertEqual(report['recorded_feature_representation'], 'legacy_filled_consensus')
        self.assertEqual(report['feature_representation'], replay.BOUNDARY_REPRESENTATION)
        self.assertFalse(report['historical_state_equivalence_required'])

    def test_recorded_boundary_replays_without_candidate_override(self):
        self.fixture['feature_representation'] = replay.BOUNDARY_REPRESENTATION
        observation, extract = self.extractor(version=1)
        report = self.execute(motion_factor=True)
        self.assert_boundary_call(observation, extract)
        self.assertFalse(report['candidate_boundary_override'])
        self.assertTrue(report['historical_state_equivalence_required'])
        self.assertEqual(report['recorded_boundary_version'],1)
        self.assertEqual(report['effective_boundary_version'],1)

    def test_graph_v2_recording_requires_exact_declared_source_contract(self):
        self.fixture.update(feature_representation=replay.BOUNDARY_REPRESENTATION,boundary_version=2)
        observation,extract=self.extractor()
        report=self.execute(motion_factor=True)
        self.assert_boundary_call(observation,extract)
        self.assertEqual(report['effective_boundary_version'],2)
        self.assertFalse(report['changed_model'])

    def test_v1_recording_cannot_silently_use_current_graph_source(self):
        self.fixture['feature_representation']=replay.BOUNDARY_REPRESENTATION
        self.extractor()
        with self.assertRaisesRegex(ValueError,'differs from selected source'):
            self.execute(motion_factor=True,candidate_evaluation=True)
        self.engine.process.assert_not_called()

    def test_v1_to_graph_requires_explicit_model_override_and_reports_both(self):
        self.fixture['feature_representation']=replay.BOUNDARY_REPRESENTATION
        self.extractor()
        report=self.execute(motion_factor=True,candidate_evaluation=True,candidate_boundary=True)
        self.assertEqual(report['recorded_boundary_version'],1)
        self.assertEqual(report['candidate_boundary_version'],2)
        self.assertTrue(report['changed_model'])

    def test_graph_v2_recording_rejects_old_source(self):
        self.fixture.update(feature_representation=replay.BOUNDARY_REPRESENTATION,boundary_version=2)
        self.extractor(version=1)
        with self.assertRaisesRegex(ValueError,'differs from selected source'):
            self.execute(motion_factor=True)
        self.engine.process.assert_not_called()

    def test_declared_graph_version_requires_topology_identity(self):
        self.fixture.update(feature_representation=replay.BOUNDARY_REPRESENTATION,boundary_version=2)
        self.extractor()
        self.modules['lane_mapping_lya_reference.paint_geometry'].TOPOLOGY_REPRESENTATION='invented-nearest-neighbours'
        with self.assertRaisesRegex(ValueError,'topology contract'):
            self.execute(motion_factor=True)
        self.engine.process.assert_not_called()

    def test_unknown_bool_or_orphan_version_is_rejected_before_core(self):
        for version in (True,0,4,None,'2'):
            self.fixture.update(feature_representation=replay.BOUNDARY_REPRESENTATION,boundary_version=version)
            with self.subTest(version=version):
                self.assert_invalid_before_core(dict(motion_factor=True),'boundary version')
        self.fixture.pop('feature_representation')
        self.fixture['boundary_version']=2
        self.assert_invalid_before_core(dict(motion_factor=True),'requires explicit feature')

    def test_missing_or_invalid_optins_reject_before_loading(self):
        for options in ({'candidate_boundary': True},
                        {'candidate_boundary': True, 'motion_factor': True},
                        {'candidate_boundary': True, 'candidate_evaluation': True},
                        {'candidate_boundary': 1, 'candidate_evaluation': True, 'motion_factor': True}):
            with self.subTest(options=options):
                self.assert_invalid_before_core(options, 'candidate boundary requires')

    def test_candidate_cannot_reinterpret_schema2(self):
        self.fixture.update(schema_version=2, kind='observed_independent_motion_mask_prefix')
        self.assert_invalid_before_core(dict(candidate_boundary=True, candidate_evaluation=True,
                                            motion_factor=True), 'schema3')

    def test_recorded_unknown_mode_is_not_legacy_fallback(self):
        self.fixture['feature_representation'] = 'unknown'
        self.assert_invalid_before_core(dict(motion_factor=True), 'boundary representation')

    def test_extractor_failure_does_not_fallback_to_filled(self):
        _, extract = self.extractor()
        extract.side_effect = ValueError('invalid real boundary observation')
        with self.assertRaisesRegex(ValueError, 'invalid real boundary'):
            self.execute(motion_factor=True, candidate_evaluation=True, candidate_boundary=True)
        self.engine.process.assert_not_called()

    def test_awaiting_transport_resolves_before_boundary_export_guard(self):
        rows = [dict(event='startup', motion_transport='awaiting_motion_protocol',
                     provenance={'feature_representation': replay.BOUNDARY_REPRESENTATION}),
                dict(event='motion', protocol=IMU_PROTOCOL)]
        (self.directory/'events.jsonl').write_text('\n'.join(map(json.dumps, rows)))
        with patch.object(replay, 'export_independent_fixture', return_value='dispatched') as export:
            self.assertEqual(replay.export_fixture(self.directory, self.directory,
                self.directory/'out.json', self.directory), 'dispatched')
        export.assert_called_once()

    def test_boundary_export_rejects_legacy_motion_protocol(self):
        rows = [dict(event='startup', motion_transport='awaiting_motion_protocol',
                     provenance={'feature_representation': replay.BOUNDARY_REPRESENTATION}),
                dict(event='motion', protocol=PROTOCOL)]
        (self.directory/'events.jsonl').write_text('\n'.join(map(json.dumps, rows)))
        with patch.object(replay, 'export_independent_fixture') as export, self.assertRaisesRegex(ValueError, 'schema3'):
            replay.export_fixture(self.directory, self.directory, self.directory/'out.json', self.directory)
        export.assert_not_called()

    def export_boundary_fixture(self, version, source_version):
        plane.PlaneReplayRouting.export_camera(self,True)
        path=self.directory/'events.jsonl'
        rows=[json.loads(line) for line in path.read_text().splitlines()]
        self.hashes={replay.source_path(name):'unchanged-source' for name in replay.BOUNDARY_CORE_NAMES}
        rows[0]['provenance'].update(feature_representation=replay.BOUNDARY_REPRESENTATION,
                                    boundary_version=version,source_sha256=self.hashes)
        path.write_text('\n'.join(map(json.dumps,rows)))
        self.extractor(version=source_version)
        output=self.directory/'boundary_export.json'
        with patch.object(replay,'load_core',return_value=(self.modules,self.hashes,{},None)), \
                patch.object(replay,'RecordedIndependentMotion',return_value=SimpleNamespace(consume=lambda _:self.sample)), \
                patch.object(replay,'verify_raw_imu_frame'):
            replay.export_fixture(self.directory,self.directory,output,self.directory)
        return json.loads(output.read_text())

    def test_export_preserves_explicit_graph_version_and_complete_sources(self):
        fixture=self.export_boundary_fixture(2,2)
        self.assertEqual(fixture['boundary_version'],2)
        self.assertEqual(fixture['feature_representation'],replay.BOUNDARY_REPRESENTATION)
        self.assertEqual(fixture['recorded_source_sha256'],self.hashes)
        self.assertEqual(fixture['camera'],plane.camera(True))

    def test_export_rejects_version_source_disagreement(self):
        with self.assertRaisesRegex(ValueError,'differs from export verification source'):
            self.export_boundary_fixture(1,2)
        self.assertFalse((self.directory/'boundary_export.json').exists())


class BoundaryCli(unittest.TestCase):
    def invoke(self, flags):
        argv = ['replay_observed_keyframe.py', 'replay', '--fixture', 'fixture.json',
                '--repo', 'repo', '--output', 'nonexistent-boundary-test-report.json'] + flags
        with patch.object(sys, 'argv', argv), patch.object(replay, 'replay_fixture', return_value={}) as call, \
                contextlib.redirect_stdout(io.StringIO()), contextlib.redirect_stderr(io.StringIO()):
            replay.main()
        return call.call_args.kwargs

    def test_legacy_call_keeps_two_original_kwargs(self):
        self.assertEqual(self.invoke([]), {'motion_factor': False, 'candidate_evaluation': False})

    def test_boundary_requires_both_explicit_flags(self):
        for flags in ([], ['--motion-factor'], ['--candidate-evaluation']):
            with self.subTest(flags=flags), self.assertRaises(SystemExit) as error:
                self.invoke(flags + ['--candidate-boundary'])
            self.assertEqual(error.exception.code, 2)

    def test_explicit_cli_routes_new_keyword_only_when_requested(self):
        self.assertEqual(self.invoke(['--candidate-boundary', '--motion-factor', '--candidate-evaluation']),
                         {'motion_factor': True, 'candidate_evaluation': True, 'candidate_boundary': True})


if __name__ == '__main__':
    unittest.main()
